       ```csharp
       using Microsoft.EntityFrameworkCore.Diagnostics;
       using System.Data.Common;
       using System.Threading;

       namespace CrowdQR.Api.Tests.Helpers;

       /// <summary>
       /// EF Core interceptor that counts database commands for testing query budgets.
       /// </summary>
       public class QueryCountingInterceptor : DbCommandInterceptor
       {
           private int _queryCount;

           /// <summary>
           /// Gets the current query count since last reset.
           /// </summary>
           public int QueryCount => _queryCount;

           /// <summary>
           /// Resets the query counter to zero.
           /// </summary>
           public void Reset() => Interlocked.Exchange(ref _queryCount, 0);

           public override InterceptionResult<DbDataReader> ReaderExecuting(
               DbCommand command, CommandEventData eventData, InterceptionResult<DbDataReader> result)
           {
               Interlocked.Increment(ref _queryCount);
               return base.ReaderExecuting(command, eventData, result);
           }

           public override ValueTask<InterceptionResult<DbDataReader>> ReaderExecutingAsync(
               DbCommand command, CommandEventData eventData, InterceptionResult<DbDataReader> result,
               CancellationToken cancellationToken = default)
           {
               Interlocked.Increment(ref _queryCount);
               return base.ReaderExecutingAsync(command, eventData, result, cancellationToken);
           }

           public override InterceptionResult<int> NonQueryExecuting(
               DbCommand command, CommandEventData eventData, InterceptionResult<int> result)
           {
               Interlocked.Increment(ref _queryCount);
               return base.NonQueryExecuting(command, eventData, result);
           }

           public override ValueTask<InterceptionResult<int>> NonQueryExecutingAsync(
               DbCommand command, CommandEventData eventData, InterceptionResult<int> result,
               CancellationToken cancellationToken = default)
           {
               Interlocked.Increment(ref _queryCount);
               return base.NonQueryExecutingAsync(command, eventData, result, cancellationToken);
           }

           public override InterceptionResult<object> ScalarExecuting(
               DbCommand command, CommandEventData eventData, InterceptionResult<object> result)
           {
               Interlocked.Increment(ref _queryCount);
               return base.ScalarExecuting(command, eventData, result);
           }

           public override ValueTask<InterceptionResult<object>> ScalarExecutingAsync(
               DbCommand command, CommandEventData eventData, InterceptionResult<object> result,
               CancellationToken cancellationToken = default)
           {
               Interlocked.Increment(ref _queryCount);
               return base.ScalarExecutingAsync(command, eventData, result, cancellationToken);
           }
       }
       ```

    ```csharp
    namespace CrowdQR.Api.Tests.Integration;

    /// <summary>
    /// Integration tests verifying query count budgets for critical API endpoints.
    /// These tests detect N+1 regressions and ensure query optimizations remain effective.
    /// </summary>
    public class QueryCountTests : BaseIntegrationTest
    {
        // Query budget thresholds per endpoint type:
        // - Simple GET by ID: 1 query
        // - GET with related data: 1-2 queries
        // - Dashboard summary: 2-3 queries
        // - Report generation: 2-3 queries

        [Fact]
        public async Task GetEvent_WithRequestsAndVotes_StaysUnderQueryBudget()
        {
            // Arrange: Seed an event with requests and votes
            await SeedStandardTestDataAsync();
            ResetQueryCount();

            // Act: Call GET /api/event/{id}
            var response = await Client.GetAsync("/api/event/1");

            // Assert: Query count under budget (1-2 queries for event + requests with vote counts)
            var queryCount = GetQueryCount();
            response.StatusCode.Should().Be(System.Net.HttpStatusCode.OK);
            queryCount.Should().BeLessOrEqualTo(2,
                $"GetEvent should use 1-2 queries (projection with subquery), but used {queryCount}");
        }

        [Fact]
        public async Task GetRequestsByEvent_WithVotes_StaysUnderQueryBudget()
        {
            // Arrange
            await SeedStandardTestDataAsync();
            ResetQueryCount();

            // Act
            var response = await Client.GetAsync("/api/request/event/1");

            // Assert
            var queryCount = GetQueryCount();
            response.StatusCode.Should().Be(System.Net.HttpStatusCode.OK);
            queryCount.Should().BeLessOrEqualTo(2,
                $"GetRequestsByEvent should use 1-2 queries, but used {queryCount}");
        }

        [Fact]
        public async Task GetEventSummary_WithRequestsAndSessions_StaysUnderQueryBudget()
        {
            // Arrange
            await SeedStandardTestDataAsync();
            // Need auth token for DJ endpoint
            var token = await GetDjAuthToken();
            ResetQueryCount();

            // Act
            var request = new HttpRequestMessage(HttpMethod.Get, "/api/dashboard/event/1/summary");
            request.Headers.Authorization = new System.Net.Http.Headers.AuthenticationHeaderValue("Bearer", token);
            var response = await Client.SendAsync(request);

            // Assert: Dashboard summary uses 2-3 queries (exists check + requests + sessions)
            var queryCount = GetQueryCount();
            response.StatusCode.Should().Be(System.Net.HttpStatusCode.OK);
            queryCount.Should().BeLessOrEqualTo(3,
                $"GetEventSummary should use 2-3 queries, but used {queryCount}");
        }

        [Fact]
        public async Task GetEvents_ListEndpoint_StaysUnderQueryBudget()
        {
            // Arrange
            await SeedStandardTestDataAsync();
            ResetQueryCount();

            // Act
            var response = await Client.GetAsync("/api/event");

            // Assert: List endpoint should use 1 query (projection)
            var queryCount = GetQueryCount();
            response.StatusCode.Should().Be(System.Net.HttpStatusCode.OK);
            queryCount.Should().BeLessOrEqualTo(1,
                $"GetEvents should use 1 query (projection), but used {queryCount}");
        }

        [Fact]
        public async Task GetVotesByRequest_StaysUnderQueryBudget()
        {
            // Arrange
            await SeedStandardTestDataAsync();
            // Need auth for vote endpoints
            var token = await GetDjAuthToken();
            ResetQueryCount();

            // Act
            var request = new HttpRequestMessage(HttpMethod.Get, "/api/vote/request/1");
            request.Headers.Authorization = new System.Net.Http.Headers.AuthenticationHeaderValue("Bearer", token);
            var response = await Client.SendAsync(request);

            // Assert
            var queryCount = GetQueryCount();
            response.StatusCode.Should().Be(System.Net.HttpStatusCode.OK);
            queryCount.Should().BeLessOrEqualTo(1,
                $"GetVotesByRequest should use 1 query, but used {queryCount}");
        }
    }
    ```

    Notes:
    - `GetDjAuthToken()` is a helper method that should be added to BaseIntegrationTest if not already present. It calls POST /api/auth/login with DJ credentials and returns the JWT token.
    - `SeedStandardTestDataAsync()` should seed enough data to expose N+1 patterns: at least 1 event with 5+ requests, each with 2+ votes.
    - Query budgets are intentionally generous (allow 1-2 extra) to avoid flaky tests while still catching N+1 patterns (which would be 10+ queries)
    - If auth is needed for some endpoints, seed a DJ user and get a token first. The auth calls themselves should NOT be counted (reset counter after auth).
